1 · Hash map — Two Sum (LC 1)

Idea: for each number, check whether its partner (target − x) was seen. O(n).

def two_sum(nums, t):
    seen = {}
    for i, x in enumerate(nums):
        if t - x in seen:
            return [seen[t - x], i]
        seen[x] = i
# two_sum([2,7,11,15], 9) -> [0, 1]

Also: group anagrams (LC 49) — key = sorted word; top-k frequent (LC 347) — Counter(nums).most_common(k).

2 · Running value — Best time to buy and sell stock (LC 121)

Track the lowest price so far; best profit = max(price − lowest). O(n).

def max_profit(p):
    lo, best = float('inf'), 0
    for x in p:
        lo = min(lo, x)
        best = max(best, x - lo)
    return best
# [7,1,5,3,6,4] -> 5

Same family: maximum subarray (Kadane, LC 53): cur = max(x, cur + x) → [−2,1,−3,4,−1,2,1,−5,4] gives 6.

3 · Two pointers — Valid palindrome (LC 125) 
def is_pal(s):
    s = [c.lower() for c in s if c.isalnum()]
    i, j = 0, len(s) - 1
    while i < j:
        if s[i] != s[j]: return False
        i += 1; j -= 1
    return True
4 · Sliding window — Longest substring without repeats (LC 3)

Move the window start past the last repeat of the current character. O(n).

def longest_unique(s):
    last, start, best = {}, 0, 0
    for i, c in enumerate(s):
        if c in last and last[c] >= start:
            start = last[c] + 1
        last[c] = i
        best = max(best, i - start + 1)
    return best
# "abcabcbb" -> 3
5 · Stack — Valid parentheses (LC 20) 
def valid_paren(s):
    st, m = [], {')': '(', ']': '[', '}': '{'}
    for c in s:
        if c in m:
            if not st or st.pop() != m[c]: return False
        else:
            st.append(c)
    return not st
6 · Binary search (LC 704) 
def bsearch(a, t):
    lo, hi = 0, len(a) - 1
    while lo <= hi:
        mid = (lo + hi) // 2
        if a[mid] == t: return mid
        if a[mid] < t: lo = mid + 1
        else: hi = mid - 1
    return -1

Python shortcut: bisect.bisect_left(a, t). Binary search also works on the answer ("smallest capacity that ships in D days").

7 · BFS on a grid — Number of islands (LC 200) 
from collections import deque
def islands(grid):
    R, C, seen, n = len(grid), len(grid[0]), set(), 0
    for r in range(R):
        for c in range(C):
            if grid[r][c] == '1' and (r, c) not in seen:
                n += 1; q = deque([(r, c)]); seen.add((r, c))
                while q:
                    a, b = q.popleft()
                    for da, db in ((1,0), (-1,0), (0,1), (0,-1)):
                        x, y = a + da, b + db
                        if 0 <= x < R and 0 <= y < C and grid[x][y] == '1' and (x, y) not in seen:
                            seen.add((x, y)); q.append((x, y))
    return n
8 · Simple DP — Climbing stairs (LC 70) and coin change (LC 322) 
def climb(n):           # ways to climb n steps, 1 or 2 at a time
    a, b = 1, 1
    for _ in range(n): a, b = b, a + b
    return a            # climb(5) -> 8

def coin_change(coins, amt):   # fewest coins
    dp = [0] + [float('inf')] * amt
    for v in range(1, amt + 1):
        for c in coins:
            if c <= v: dp[v] = min(dp[v], dp[v - c] + 1)
    return dp[amt] if dp[amt] != float('inf') else -1
# coin_change([1,2,5], 11) -> 3

DP recipe: define the state (dp[v] = best answer for v), the transition, the base case, the order.
